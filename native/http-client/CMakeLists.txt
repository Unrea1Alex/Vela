cmake_minimum_required(VERSION 3.20)
project(vela_http_client C)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Vela HTTP multi adapter requires Windows x64 MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_MSVC_STATIC_RT TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED)
find_package(CURL 8.23.0 EXACT CONFIG REQUIRED)

function(add_pool target)
    add_library(${target} STATIC pool.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PUBLIC CURL::libcurl)
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()
function(add_probe target pool)
    add_executable(${target} ../../tests/library13/http-multi-probe.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PRIVATE ${pool})
endfunction()

add_pool(vela_http_pool)
add_library(vela-http SHARED adapter.c)
target_compile_features(vela-http PRIVATE c_std_11)
target_compile_options(vela-http PRIVATE /W4 /WX /utf-8)
target_link_libraries(vela-http PRIVATE vela_http_pool)
add_probe(vela-http-probe vela_http_pool)
add_pool(vela_http_pool_faults)
target_compile_definitions(vela_http_pool_faults PUBLIC VELA_HTTP_TEST_FAULTS=1)
add_probe(vela-http-fault-probe vela_http_pool_faults)
